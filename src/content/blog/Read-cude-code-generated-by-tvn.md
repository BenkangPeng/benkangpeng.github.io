> TVM生成的cuda代码往往带有大量抽象的索引计算，可读性较差。这篇blog将分析如何更好地读懂cuda code generated by TVM

以下是一个`4096x4096x4096 GEMM`的`IRModule`. `grid size = <512,1,1>` ，一个`threadblock`负责计算C矩阵一个`256x128`的分块； ` block size = <256,1,1>`，一个`thread`负责计算一个`16x8`的分块。各变量大小：

* `BK=8`    `ax0_0`循环512次，说明一个`threadblock`需要512次迭代才能计算完`256x4096x128`的矩阵乘，所以每次迭代计算大小为`256x8x128`的矩阵乘。
* `A_shared:256x8=2048 `
* `B_shared: 8x128=1024`
* `C_local: 128`  

```python
# from tvm.script import ir as I
# from tvm.script import tir as T

@I.ir_module
class Module:
    @T.prim_func
    def main(A: T.Buffer((4096, 4096), "float32"), B: T.Buffer((4096, 4096), "float32"), compute: T.Buffer((4096, 4096), "float32")):
        T.func_attr({"tir.noalias": True})
        # with T.block("root"):
        A_shared = T.alloc_buffer((4096, 4096), scope="shared")
        B_shared = T.alloc_buffer((4096, 4096), scope="shared")
        compute_local = T.alloc_buffer((4096, 4096), scope="local")
        for ax0_0_ax1_0_fused in T.thread_binding(512, thread="blockIdx.x"):
            for ax0_1 in T.thread_binding(16, thread="vthread.x"):
                for ax1_1 in T.thread_binding(8, thread="vthread.y"):
                    for ax0_2_ax1_2_fused_init in T.thread_binding(256, thread="threadIdx.x"):
                        with T.block("compute_init"):
                            v_y = T.axis.spatial(4096, ax0_0_ax1_0_fused // 32 * 256 + ax0_1 * 16 + ax0_2_ax1_2_fused_init // 16)
                            v_x = T.axis.spatial(4096, ax0_0_ax1_0_fused % 32 * 128 + ax1_1 * 16 + ax0_2_ax1_2_fused_init % 16)
                            T.reads()
                            T.writes(compute_local[v_y, v_x])
                            compute_local[v_y, v_x] = T.float32(0.0)
                    for ax0_2_ax1_2_fused in T.thread_binding(256, thread="threadIdx.x"):
                        for ax0_0 in range(512):
                            for ax0_ax1_fused_0_0 in T.unroll(2):
                                for ax0_ax1_fused_0_1 in T.thread_binding(256, thread="threadIdx.x"):
                                    for ax0_ax1_fused_1 in T.vectorized(4):
                                        with T.block("A_shared"):
                                            v0 = T.axis.spatial(4096, ax0_0_ax1_0_fused // 32 * 256 + (ax0_ax1_fused_0_0 * 1024 + ax0_ax1_fused_0_1 * 4 + ax0_ax1_fused_1) // 8)
                                            v1 = T.axis.spatial(4096, ax0_0 * 8 + (ax0_ax1_fused_0_0 * 1024 + ax0_ax1_fused_0_1 * 4 + ax0_ax1_fused_1) % 8)
                                            T.reads(A[v0, v1])
                                            T.writes(A_shared[v0, v1])
                                            A_shared[v0, v1] = A[v0, v1]
                            for ax0_ax1_fused_0_0 in T.unroll(1):
                                for ax0_ax1_fused_0_1 in T.thread_binding(256, thread="threadIdx.x"):
                                    for ax0_ax1_fused_1 in T.vectorized(4):
                                        with T.block("B_shared"):
                                            v0 = T.axis.spatial(4096, ax0_0 * 8 + (ax0_ax1_fused_0_0 * 1024 + ax0_ax1_fused_0_1 * 4 + ax0_ax1_fused_1) // 128)
                                            v1 = T.axis.spatial(4096, ax0_0_ax1_0_fused % 32 * 128 + (ax0_ax1_fused_0_0 * 1024 + ax0_ax1_fused_0_1 * 4 + ax0_ax1_fused_1) % 128)
                                            T.reads(B[v0, v1])
                                            T.writes(B_shared[v0, v1])
                                            B_shared[v0, v1] = B[v0, v1]
                            for ax0_1_1, ax0_2 in T.grid(8, 1):
                                with T.block("compute_update"):
                                    v_y = T.axis.spatial(4096, ax0_0_ax1_0_fused // 32 * 256 + ax0_1 * 16 + ax0_2_ax1_2_fused // 16)
                                    v_x = T.axis.spatial(4096, ax0_0_ax1_0_fused % 32 * 128 + ax1_1 * 16 + ax0_2_ax1_2_fused % 16)
                                    v_k = T.axis.reduce(4096, ax0_0 * 8 + ax0_1_1 + ax0_2)
                                    T.reads(compute_local[v_y, v_x], A_shared[v_y, v_k], B_shared[v_k, v_x])
                                    T.writes(compute_local[v_y, v_x])
                                    compute_local[v_y, v_x] = compute_local[v_y, v_x] + A_shared[v_y, v_k] * B_shared[v_k, v_x]
                        for ax0_3, ax1_3 in T.grid(1, 1):
                            with T.block("compute_local"):
                                v0 = T.axis.spatial(4096, ax0_0_ax1_0_fused // 32 * 256 + ax0_1 * 16 + ax0_2_ax1_2_fused // 16 + ax0_3)
                                v1 = T.axis.spatial(4096, ax0_0_ax1_0_fused % 32 * 128 + ax1_1 * 16 + ax0_2_ax1_2_fused % 16 + ax1_3)
                                T.reads(compute_local[v0, v1])
                                T.writes(compute[v0, v1])
                                compute[v0, v1] = compute_local[v0, v1]
```

分段解读TVM生成的Cuda code：

* `C_local Init`

  8个`for loop`初始化`compute_local`，每个`for loop`中，迭代之间地址差8,可以判断`C_local`列数为8；`for loop extent`为16，判断`C_local`行数为16. 因此，每个`for loop`初始化`C_local`的一列，一列16个数，形状为`16x8`

```c++
extern "C" __global__ void __launch_bounds__(256) main_kernel(float* __restrict__ A, float* __restrict__ B, float* __restrict__ compute) {
  float compute_local[128];
  __shared__ float A_shared[2048];
  __shared__ float B_shared[1024];
  for (int vthread_x_s = 0; vthread_x_s < 16; ++vthread_x_s) {
    compute_local[(vthread_x_s * 8)] = 0.000000e+00f;
  }
  for (int vthread_x_s_1 = 0; vthread_x_s_1 < 16; ++vthread_x_s_1) {
    compute_local[((vthread_x_s_1 * 8) + 1)] = 0.000000e+00f;
  }
  for (int vthread_x_s_2 = 0; vthread_x_s_2 < 16; ++vthread_x_s_2) {
    compute_local[((vthread_x_s_2 * 8) + 2)] = 0.000000e+00f;
  }
  for (int vthread_x_s_3 = 0; vthread_x_s_3 < 16; ++vthread_x_s_3) {
    compute_local[((vthread_x_s_3 * 8) + 3)] = 0.000000e+00f;
  }
  for (int vthread_x_s_4 = 0; vthread_x_s_4 < 16; ++vthread_x_s_4) {
    compute_local[((vthread_x_s_4 * 8) + 4)] = 0.000000e+00f;
  }
  for (int vthread_x_s_5 = 0; vthread_x_s_5 < 16; ++vthread_x_s_5) {
    compute_local[((vthread_x_s_5 * 8) + 5)] = 0.000000e+00f;
  }
  for (int vthread_x_s_6 = 0; vthread_x_s_6 < 16; ++vthread_x_s_6) {
    compute_local[((vthread_x_s_6 * 8) + 6)] = 0.000000e+00f;
  }
  for (int vthread_x_s_7 = 0; vthread_x_s_7 < 16; ++vthread_x_s_7) {
    compute_local[((vthread_x_s_7 * 8) + 7)] = 0.000000e+00f;
  }
```

* `Load A/B into A/B_shared`

`ax0_0 for loop extent = 512`，表示一个`threadblock`要用512次迭代计算完256x4096x128，每次迭代计算`256x8x128`

重点来看B的数据搬运：采用float4一次搬运4个float，256个threads一起将`8x128`的B搬到`B_shared`；

```cpp
for (int ax0_0 = 0; ax0_0 < 512; ++ax0_0) {
    __syncthreads();
    *(float4*)(A_shared + (((int)threadIdx.x) * 4)) = *(float4*)(A + (((((((int)blockIdx.x) >> 5) * 1048576) + ((((int)threadIdx.x) >> 1) * 4096)) + (ax0_0 * 8)) + ((((int)threadIdx.x) & 1) * 4)));
    *(float4*)(A_shared + ((((int)threadIdx.x) * 4) + 1024)) = *(float4*)(A + ((((((((int)blockIdx.x) >> 5) * 1048576) + ((((int)threadIdx.x) >> 1) * 4096)) + (ax0_0 * 8)) + ((((int)threadIdx.x) & 1) * 4)) + 524288));
    *(float4*)(B_shared + (((int)threadIdx.x) * 4)) = *(float4*)(B + ((((ax0_0 * 32768) + ((((int)threadIdx.x) >> 5) * 4096)) + ((((int)blockIdx.x) & 31) * 128)) + ((((int)threadIdx.x) & 31) * 4)));
    __syncthreads();
```



```cpp
    *(float4*)(B_shared + (((int)threadIdx.x) * 4)) = *(float4*)(B + ((((ax0_0 * 32768) + ((((int)threadIdx.x) >> 5) * 4096)) + ((((int)blockIdx.x) & 31) * 128)) + ((((int)threadIdx.x) & 31) * 4)));
```

* `B_shared + threadIdx.x * 4` 找到每一个线程写入`B_shared`的起始地址
* `blockIdx.x & 31 * 128`  `&31`即`%32`, 找到当前`threadblock`计算的大小为`4096x128`的B分块的起始地址(左上角)，`*128`是指相邻`B block`起始地址相差128

* `ax0_0 * 32768 = ax0_0 * 8 * 4096` 找到当前迭代需要计算的大小为`8x128`的B分块的起始地址；相邻B(小)分块之间相差`8 * 4096`个数
* `threadIdx.x >> 5 * 4096` 在单个`8x128`的block中，当前线程之上，有多少行线程，每行4096个数
* `threadIdx.x & 31 * 4` 当前线程在当前行中，之前有多少个线程，线程与线程间加载数据的起始位置相差4
